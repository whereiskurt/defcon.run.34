**3. Update `apps/run.cms/nginx/nginx.conf`:**

Add TWO new location blocks BEFORE the existing `location /` fallback block (order matters):

```nginx
# Serve branded login page at exact root
location = / {
  root /etc/nginx/html;
  index index.html;
}

# Serve static assets (background images)
location /bg/ {
  root /etc/nginx/html;
  expires 30d;
  add_header Cache-Control "public, immutable";
```

The `location = /` (exact match) takes priority over `location /` (prefix match), so the root URL serves the branded page while all other unprefixed paths still proxy to Strapi. **1. Add a `renderBrandedError` function** at the top of the file (after imports, before `REQUIRED_SERVICE` constant):

```typescript
function escapeHtml(str: string): string {
  return str
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}

function renderBrandedError(title: string, message: string, actions: string): string {

  <style>
    /* EXACT values from auth.defcon.run - see globals.css and tailwind.config.js */
    * { box-sizing: border-box; margin: 0; padding: 0; }
    body { background: #0a0a0f; color: #e4e4e7; font-family: 'MuseoModerno', 'Segoe UI', system-ui, sans-serif; min-height: 100vh; display: flex; align-items: center; justify-content: center; overflow: hidden; }
```
